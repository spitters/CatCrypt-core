-- Counted by `point_mass` at the definition and by `point_mass_use` at the theorem.

-- Counted by `free_witness`: body and frame are existential.
theorem fx_free_both (h : compile m = some (P, bytes)) :
    ∃ body frame fuel s, runA frame body fuel e = some s ∧ haltedA body s = true
      ∧ RunSafe frame body (AccessIn (layoutMem σ sa) frame body) fuel e false := proof h

-- Counted by `free_witness`: the body is compiled, the frame map is existential.
theorem fx_free_frame :
    ∃ body, decode bytes = some body ∧ ∃ frame, AccessIn mem frame body x := proof

-- Counted by `free_witness`: a predicate whose body is an existential conjunct.
def FxSafe (e : AState) : Prop :=
  ∃ body, StoreIn mem fr body e

-- Not counted: both are functions of the compiled bytes.
theorem fx_tied_both :
    ∃ body frame, decode bytes = some (body, frame)
      ∧ RunSafe frame body (AccessIn mem frame body) fuel e false := proof

-- Not counted: an equation gives the frame, the body is an argument.
theorem fx_tied_eq (body : List Instr) :
    ∃ frame, frame = frameOf σ ∧ LoadIn mem frame body x := proof

-- Not counted: only the number of steps is existential.
theorem fx_tied_fuel :
    ∃ fuel, RunSafe (frameOf σ) (bodyOf bytes) I fuel e false := proof

-- Counted by `returned_arg`: the witness is the theorem's argument.
theorem fx_returned (p : Prog) (ctx : Ctx) :
    ∃ src : Ctx, trace p src = trace p ctx :=
  ⟨ctx, rfl⟩

-- Counted by `returned_arg`: the same through tactics.
theorem fx_returned_tac (p : Prog) : ∀ (ctx : Ctx), ∃ src, trace p src = trace p ctx := by
  intro ctx
  exact ⟨ctx, fun _ => rfl⟩

-- Counted by `returned_arg`: the same through `use`.
theorem fx_returned_use (p : Prog) (ctx : Ctx) : ∃ src, trace p src = trace p ctx := by
  use ctx

-- Not counted: the witness is a function of the argument.
theorem fx_translated (p : Prog) (ctx : Ctx) :
    ∃ src : Ctx, trace p src = trace (compile p) ctx :=
  ⟨backTranslate ctx, rfl⟩

-- Not counted: the argument comes with a proof.
theorem fx_witnessed (n : Nat) (hn : 0 < n) : ∃ k, 0 < k :=
  ⟨n, hn⟩

-- Not counted: no existential conclusion.
theorem fx_plain (ctx : Ctx) : trace p ctx = trace p ctx := rfl

-- Counted by `free_witness`: an equation of the body with itself determines nothing.
theorem fx_free_self :
    ∃ body, body = body ∧ RunSafe fr body I fuel e false := proof

-- Not counted: the equation that gives the body is in a binder.

    ∃ (body : List Instr) (_ : decode bytes = some body), RunSafe fr body I fuel e false := proof
